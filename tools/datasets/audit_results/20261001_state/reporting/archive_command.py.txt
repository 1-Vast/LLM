from pathlib import Path
import hashlib
import json
import shutil

root = Path('D:/MAESTRO')
run = root / 'outputs/state_identifiability_20261001'

def sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(4 << 20), b''):
            h.update(block)
    return h.hexdigest()

def write(path, value):
    with path.open('x', encoding='utf-8') as handle:
        json.dump(value, handle, indent=2)
        handle.write('\n')

source_paths = [
    'tools/datasets/state_identifiability.py', 'tools/datasets/state_raw_review.py',
    'tools/datasets/state_public_review.py', 'tools/datasets/state_public_sources_20261001.json',
    'tools/datasets/catalog.py', 'research/identifiability_audit/lineage_sciplex3.py',
    'research/identifiability_audit/lineage_l1000.py', 'research/biological_depth/prepare.py',
    'research/dual_core_completion/verify.py', 'research/dual_core_followup/snapshot.py',
    'tests/test_repository_shape.py', 'tests/fixtures/historical_markdown.json',
    'tests/test_state_identifiability.py', 'tests/test_state_public_review.py',
    'tests/test_learned_response.py', 'pyproject.toml',
]
source_out = run / 'execution_sources'
source_out.mkdir(exist_ok=False)
sources = {}
for relative in source_paths:
    original = root / relative
    target = source_out / (relative + '.txt')
    target.parent.mkdir(parents=True, exist_ok=True)
    shutil.copyfile(original, target)
    digest = sha(original)
    assert sha(target) == digest
    sources[relative] = {'sha256': digest, 'archived_path': target.relative_to(run).as_posix()}
write(source_out / 'source_manifest.json', sources)

write(run / 'full_verification/test_scope.json', {
    'result': '1659 passed; 0 failures, errors or skips; 2 deselected',
    'excluded_nodeids': [
        'tests/test_learned_response.py::test_small_response_fit_learns_and_restores_selected_epoch',
        'tests/test_learned_response.py::test_training_refuses_nonfinite_values',
    ],
    'exclusion_semantics': ['constructed small-model fitting', 'fitting-entry refusal, no successful fit'],
    'historical_language_isolation': json.loads((root / 'tests/fixtures/historical_markdown.json').read_text()),
    'scope': 'engineering contracts with constructed fixtures; no scientific state-gain arm, no new biological fitting',
    'validation_sha256': sha(run / 'full_verification/validation.json'),
})

# No credentials were used by this scientific audit. Safely check configured
# secret values before publication without displaying their names or values.
secrets = []
env = root / '.env'
if env.is_file():
    for line in env.read_text(encoding='utf-8').splitlines():
        if '=' not in line or line.lstrip().startswith('#'):
            continue
        name, value = line.split('=', 1)
        value = value.strip().strip('"').strip("'")
        if ('KEY' in name.upper() or 'TOKEN' in name.upper() or 'SECRET' in name.upper()) and len(value) >= 12:
            secrets.append(value.encode())

target = root / 'outputs/state_identifiability_publication_20261001'
target.mkdir(exist_ok=False)
records = []
privacy_hits = []
for original in sorted(run.rglob('*')):
    if not original.is_file():
        continue
    relative = original.relative_to(run)
    excluded = any(part in {'test_temp', '__pycache__', '.pytest_cache'} for part in relative.parts)
    digest = sha(original)
    records.append({'original_path': original.relative_to(root).as_posix(), 'sha256': digest,
                    'published': not excluded, 'reason': 'generated_test_fixture_or_cache' if excluded else None})
    if excluded:
        continue
    content = original.read_bytes()
    if any(value in content for value in secrets):
        privacy_hits.append(relative.as_posix())
    destination = target / relative
    destination.parent.mkdir(parents=True, exist_ok=True)
    with destination.open('xb') as handle:
        handle.write(content)
    assert sha(destination) == digest
assert not privacy_hits, 'privacy review found configured secret bytes; no publication permitted'
write(target / 'curation_receipt.json', {
    'source': run.relative_to(root).as_posix(), 'records': records,
    'published_originals': sum(item['published'] for item in records),
    'excluded_local_originals': sum(not item['published'] for item in records),
    'privacy_hits': privacy_hits, 'original_files_preserved': True,
    'code_and_command': 'reporting/archive_command.py.txt, executed as PowerShell Get-Content -Raw | maestro python -',
})
print(json.dumps({'published': sum(item['published'] for item in records),
                  'excluded': sum(not item['published'] for item in records), 'privacy_hits': len(privacy_hits)}))
