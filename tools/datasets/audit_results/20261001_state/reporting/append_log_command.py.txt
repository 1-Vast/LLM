from pathlib import Path
import hashlib
import json

root = Path('D:/MAESTRO')
out = root / 'outputs/state_identifiability_20261001/reporting'
validation = json.loads((out.parent / 'full_verification/validation.json').read_text())
assert validation['passed'] and validation['counts']['failures'] == 0
path = root / 'log/20261001/README.md'
before = path.read_bytes()
marker = b'## 14. Real predecision-state identifiability gate, 2026-10-01'
assert marker not in before, 'dated section already exists; refuse duplicate append'
source = out / 'section14.md.txt'
text = source.read_text(encoding='utf-8')
result = f"**{validation['counts']['tests']:,} passed, zero failures/errors/skips, two fitting-related checks deselected**"
assert '__TEST_COUNTS__' in text
text = text.replace('__TEST_COUNTS__', result)
with (out / 'rendered_section14.md.txt').open('xb') as handle:
    handle.write(text.encode('utf-8'))
with path.open('ab') as handle:
    handle.write(text.encode('utf-8'))
after = path.read_bytes()
assert after[:len(before)] == before
receipt = {
    'command': 'Get-Content -Raw append_log_command.py.txt | maestro/python.exe -',
    'target': 'log/20261001/README.md',
    'original_prefix_bytes': len(before),
    'original_prefix_sha256': hashlib.sha256(before).hexdigest(),
    'original_prefix_preserved': True,
    'appended_section_sha256': hashlib.sha256(text.encode('utf-8')).hexdigest(),
    'result_sha256': hashlib.sha256(after).hexdigest(),
    'validation_sha256': hashlib.sha256((out.parent / 'full_verification/validation.json').read_bytes()).hexdigest(),
}
with (out / 'append_receipt.json').open('x', encoding='utf-8') as handle:
    json.dump(receipt, handle, indent=2)
    handle.write('\n')
print(json.dumps(receipt))
