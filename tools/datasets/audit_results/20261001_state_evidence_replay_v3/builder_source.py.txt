"""Independent original-log replay. Never estimates biological STATE gain."""
from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path

import numpy as np
from scipy.io import loadmat


def sha256(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def control_action(theta, reference):
    theta = np.asarray(theta, dtype=float).reshape(-1)
    if not np.all(np.isfinite(theta)) or not np.isfinite(reference):
        raise ValueError("nonfinite controller input")
    return int(np.sum(np.sin(theta - reference) * (theta < 0.4 * 2 * np.pi)) < 0)


def assess(design, evidence):
    """Positive evidence is required; unknown is never equivalent to passed."""
    common = {"legal_predecision_state", "sample_linkage", "real_endpoint",
              "complete_attempt_denominator", "qc", "matched_resource_rules",
              "interference_handled", "physical_batch_mapping"}
    designs = {
        "sister_panel": {"random_parent_split", "two_measured_actions", "matched_controls"},
        "random_policy": {"policy_randomization", "policy_arm_coverage"},
        "random_action_log": {"authenticated_propensities", "conditional_action_support"},
    }
    if design not in designs:
        raise ValueError("unknown comparison design")
    missing = sorted(k for k in common | designs[design] if evidence.get(k) is not True)
    if missing:
        return {"classification": "replay_only", "missing": missing}
    if evidence.get("costs_authenticated") is True:
        return {"classification": "comparable", "missing": []}
    if evidence.get("credible_predeclared_cost_bounds") is True:
        return {"classification": "predeclared_utility_interval_only", "missing": ["costs_authenticated"]}
    return {"classification": "replay_only", "missing": ["costs_or_credible_bounds"]}


def replay(path):
    data = loadmat(path, simplify_cells=True)
    records = data["PhaseMAT"]
    actions = np.asarray(data["inputLEVELS"]).reshape(-1)
    if len(records) != len(actions):
        raise ValueError("phase/action length mismatch")
    objects = {}
    for collection in ("trackedCELLS", "removedCELLS"):
        values = data[collection]
        if isinstance(values, dict):
            values = [values]
        for ordinal, obj in enumerate(values, 1):
            label = int(obj["LABEL"])
            if label in objects:
                raise ValueError("duplicate native object LABEL")
            objects[label] = (collection, ordinal, obj)
    rows = []
    source_hash = sha256(path)
    # main_run_exp.m initializes Theta_r=0 before the calibration block.
    reference = 0.0
    references = finite_responses = missing_responses = 0
    for index, record in enumerate(records):
        frame = index + 1  # original MATLAB row, not a wall-clock timestamp
        post_ref = np.asarray(record["Theta_r"]).reshape(-1)
        labels = np.asarray(record["trackedCELLS"]).reshape(-1).astype(int)
        row = {"source_row_matlab": frame, "source_sha256": source_hash,
               "recorded_action": int(actions[index]), "native_labels": labels.tolist(),
               "state_latest_frame": frame - 1, "response_frame": frame,
               "actuator_ack": None, "physical_batch": None, "qc": None, "cost": None}
        if frame <= int(data["initTP"]):
            row.update(stage="calibration", replayed_action=None)
        elif not len(post_ref) or not np.all(np.isfinite(post_ref)):
            row.update(stage="reference_not_saved", replayed_action=None)
        else:
            theta = np.asarray(record["stimaf"]).reshape(-1)
            if len(labels) != len(theta):
                raise ValueError("native LABEL/phase mismatch")
            row.update(stage="controller_replay", predecision_reference=reference,
                       postdecision_reference=float(post_ref[0]),
                       replayed_action=control_action(theta, reference),
                       same_row_wrong_action=control_action(theta, float(post_ref[0])))
        # Also retain native label linkage for nonfeedback records; these are references, not N.
        if frame > int(data["initTP"]):
            for label in labels:
                if label not in objects:
                    raise ValueError("unlinked native LABEL")
                _, _, obj = objects[label]
                frames = np.asarray(obj["frame"]).reshape(-1)
                fluo = np.asarray(obj["MaxGreenFluo"]).reshape(-1)
                if len(frames) != len(fluo) or frame - 1 not in frames:
                    raise ValueError("state does not link to prior native frame")
                references += 1
                loc = np.flatnonzero(frames == frame)
                valid = bool(len(loc) == 1 and np.isfinite(fluo[loc[0]]))
                finite_responses += valid
                missing_responses += not valid
        if len(post_ref) and np.all(np.isfinite(post_ref)):
            reference = float(post_ref[0])
        rows.append(row)
    eligible = [r for r in rows if r["stage"] == "controller_replay"]
    inventory = {"file": path.name, "records": len(rows), "replayable": len(eligible),
                 "matches": sum(r["recorded_action"] == r["replayed_action"] for r in eligible),
                 "same_row_mismatches": sum(r["recorded_action"] != r["same_row_wrong_action"] for r in eligible),
                 "label_references": references, "finite_current_frame_fluorescence": finite_responses,
                 "missing_current_frame_fluorescence": missing_responses,
                 "sampling_minutes_nominal": int(data["smplngTM"]),
                 "state_feature_identity_authenticated": None,
                 "missing_response_is_biological_failure": False}
    return rows, inventory


def run(raw, protocol, out):
    # Verify original bytes, pinned Git blob identity, and successful acquisition before parsing.
    inputs = []
    for folder in (raw, protocol):
        for receipt in json.loads((folder / "receipts.json").read_text()):
            path = folder / receipt["path"]
            if receipt["error"] or sha256(path) != receipt["sha256"]:
                raise ValueError("untrusted source receipt")
            if "git_blob_sha1" in receipt:
                content = path.read_bytes()
                blob = hashlib.sha1(b"blob " + str(len(content)).encode() + b"\0" + content).hexdigest()
                if blob != receipt["git_blob_sha1"]:
                    raise ValueError("pinned Git blob mismatch")
            inputs.append(receipt)
    out.mkdir(parents=True, exist_ok=False)
    (out / "builder_source.py.txt").write_bytes(Path(__file__).read_bytes())
    summaries = []
    with (out / "decisions.jsonl").open("w", encoding="utf-8") as stream:
        for path in sorted(raw.glob("*FIG*RefOsc*.raw")):
            rows, inventory = replay(path)
            summaries.append(inventory)
            for row in rows:
                stream.write(json.dumps(dict(file=path.name, **row), allow_nan=False) + "\n")
    report = {"inputs": inputs, "experiments": summaries, "model_inference": False,
              "state_efficacy_experiment": False, "classification": "replay_only",
              "boundary": "Action agreement cannot identify continuous feature equality or counterfactual support."}
    (out / "summary.json").write_text(json.dumps(report, indent=2) + "\n", encoding="utf-8")
    outputs = {p.name: sha256(p) for p in out.iterdir() if p.is_file()}
    (out / "outputs_sha256.json").write_text(json.dumps(outputs, indent=2) + "\n")
    return summaries


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--raw", type=Path, required=True)
    parser.add_argument("--protocol", type=Path, required=True)
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    print(json.dumps(run(args.raw, args.protocol, args.out), indent=2))
