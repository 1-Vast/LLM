"""Scientific boundary contracts for metadata gates and observed-only transport."""
import numpy as np
import pandas as pd
import pytest

from research.astra.gdsc_transport import exact_menu_records
from research.astra.gdsc_layout_validation import PAIR, choose_pair, normalize_scan
from research.astra.gdsc_pilot import design
from research.astra.ctrp_qualification import normalized_name


def assignment():
    return pd.DataFrame([
        dict(source_row=10, POSITION=723, DRUG_ID=1036., CONC=10., TAG="L12-D1-S", action=PAIR[0]),
        dict(source_row=11, POSITION=1155, DRUG_ID=1060., CONC=.25, TAG="R2-D1-S", action=PAIR[1])])


def native():
    rows = [dict(source_row=i, POSITION=i+1, TAG=tag, DRUG_ID=np.nan, CONC=np.nan, INTENSITY=value)
            for i,(tag,value) in enumerate([( "NC-1",100.),("NC-1",100.),("NC-1",100.),
                                             ("B",10.),("B",10.),("B",10.)])]
    rows.extend(dict(**r, INTENSITY=v) for r,v in zip(assignment().drop(columns="action").to_dict("records"),[55.,-8.]))
    return pd.DataFrame(rows)


def test_metadata_qualification_never_uses_response_or_interpolates_dose():
    x = native().assign(BARCODE=1, DRUGSET_ID=231, ASSAY="Glo", DURATION=4,
                        MASTER_CELL_ID=1, CELL_ID=2, SCAN_ID=3)
    original = exact_menu_records(x)
    changed = exact_menu_records(x.assign(INTENSITY=np.nan))
    pd.testing.assert_frame_equal(original, changed)
    x.loc[x.DRUG_ID == 1060, "CONC"] = .2501
    assert len(exact_menu_records(x)) == 1


def test_matched_shared_controls_and_unclipped_observed_values():
    rows, reasons = normalize_scan(native(), assignment())
    assert not reasons
    assert [r["utility"] for r in rows] == pytest.approx([.5,1.2])
    assert rows[0]["control_source_rows_NC1"] == rows[1]["control_source_rows_NC1"]


@pytest.mark.parametrize("failure", ["missing", "failed", "duplicate", "nonfinite"])
def test_missing_or_invalid_action_never_becomes_imputed_outcome(failure):
    x = native()
    if failure == "missing":
        x = x.loc[x.source_row != 11]
    elif failure == "failed":
        x = pd.concat([x, pd.DataFrame([dict(source_row=12,POSITION=1155,TAG="FAIL",INTENSITY=0.)])],ignore_index=True)
    elif failure == "duplicate":
        x = pd.concat([x,x.loc[x.source_row==11].assign(source_row=12)],ignore_index=True)
    else:
        x.loc[x.source_row == 11,"INTENSITY"] = np.nan
    rows, reasons = normalize_scan(x, assignment())
    assert rows == [] and reasons


def test_nonpositive_control_range_refuses_panel():
    x = native()
    x.loc[x.TAG == "NC-1", "INTENSITY"] = 10.
    rows, reasons = normalize_scan(x, assignment())
    assert not rows and "nonpositive_control_range" in reasons


def test_frozen_action_identity_mismatch_raises():
    x = native()
    x.loc[x.source_row == 11,"CONC"] = .5
    with pytest.raises(ValueError,match="identity"):
        normalize_scan(x, assignment())


def test_pair_selector_preserves_tie_rule_and_rejects_nonfinite():
    assert choose_pair([[1.,1.],[0.,1.]]).tolist() == [0,1]
    with pytest.raises(ValueError):
        choose_pair([[np.nan,1.]])


def test_pilot_has_randomized_complete_menus_and_no_fabricated_execution():
    roster = pd.DataFrame([dict(MASTER_CELL_ID=i,CELL_LINE_NAME=f"line{i}",C_action=i//4) for i in range(12)])
    blocks,wells = design(roster)
    assert len(blocks) == 24 and len(wells) == 480
    assert blocks.groupby("planned_day").size().to_dict() == {1:6,2:6,3:6,4:6}
    assert blocks.groupby("MASTER_CELL_ID").planned_day.nunique().eq(2).all()
    assert wells.planned_plate.nunique() == 8
    assert not wells.duplicated(["planned_plate","planned_well"]).any()
    assert not wells.executed.any() and wells.endpoint_value.isna().all()
    assert not blocks.independently_started_verified.any()
    assert wells.loc[wells.role == "drug"].groupby(["planned_culture_id","drug_id"]).size().eq(3).all()
    assert wells.loc[wells.role == "vehicle"].groupby("planned_culture_id").size().eq(3).all()
    assert wells.loc[wells.role == "blank"].groupby("planned_culture_id").size().eq(3).all()
    assert wells.loc[wells.role == "baseline_blank"].groupby("planned_culture_id").size().eq(3).all()
    pd.testing.assert_frame_equal(wells,design(roster)[1])


def test_pilot_rejects_incomplete_recommendation_strata():
    roster = pd.DataFrame([dict(MASTER_CELL_ID=i,CELL_LINE_NAME=f"line{i}",C_action=0) for i in range(12)])
    with pytest.raises(ValueError,match="stratum"):
        design(roster)


def test_compound_name_normalization_does_not_substitute_a_related_inhibitor():
    assert normalized_name("PD-0325901") == normalized_name("PD0325901")
    assert normalized_name("PD0325901") != normalized_name("selumetinib")
