"""Small stable research entry points reusing the supported production components.

The supplied source copies can remain independent reference material. The full
single-submission research path is DecisionPath, not this compatibility wrapper.
"""
from __future__ import annotations

import uuid

from agent.context import TaskIntent
from agent.memory import CaseState
from agent.orchestrator import MAESTROOrchestrator as ProductionOrchestrator, MAESTROTurn
from virtual_cell.state_adapter import StateAdapterConfig, StateCapabilityAdapter


class MAESTROOrchestrator(ProductionOrchestrator):
    """Return waiting cases before the inherited interpreter or model is called."""

    def run(self, user_message, *, case_id=None, budget=None, session_id=None, **kwargs):
        session_id = session_id or str(uuid.uuid4())
        case_id = case_id or session_id
        case = self._case_store.open_case(case_id, budget=budget) if self._case_store else None
        if case is not None and case.state is CaseState.AWAITING_RESULT:
            self._logger.event("case_awaiting_result", {"case_id": case_id, "plan_version": case.plan_version},
                               session_id=session_id)
            intent = TaskIntent("awaiting_result", "", (), (), None, None, (), (), (), False)
            return MAESTROTurn(session_id=session_id, intent=intent,
                               response="Import the current plan's measurements before the next decision.", case=case)
        return super().run(user_message, case_id=case_id, budget=budget, session_id=session_id, **kwargs)


__all__ = ["MAESTROOrchestrator", "StateAdapterConfig", "StateCapabilityAdapter"]
