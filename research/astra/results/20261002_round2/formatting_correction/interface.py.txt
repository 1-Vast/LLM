"""Research query identities; supported prediction types remain production-owned."""
from __future__ import annotations

import hashlib
import json
import re
from dataclasses import asdict, dataclass
from math import isfinite

from virtual_cell.interface import StatePrediction


def _finite(value):
    return not isinstance(value, bool) and isinstance(value, (int, float)) and isfinite(value)


@dataclass(frozen=True)
class StateRef:
    """Caller-bound input identity; this does not certify experiment chronology.

    In particular, available_at is an asserted reference, not proof that RNA
    processing finished before a decision. Task admission must audit that
    evidence independently. Historical endpoint controls are not predecision.
    """

    content_sha256: str
    feature_axis_sha256: str
    preprocessing_version: str
    experimental_unit: str
    role: str
    collected_at: str | None = None
    available_at: str | None = None


@dataclass(frozen=True)
class ScientificQuery:
    state: StateRef
    action_id: str
    intervention_sha256: str
    context: str
    population: str
    endpoint: str
    time_hours: float
    outcome_mode: str


@dataclass(frozen=True)
class ExecutionBinding:
    checkpoint_sha256: str
    input_transform_version: str
    sampling_indices_sha256: str
    seed: int
    runtime_version: str


def raw_prediction_key(query: ScientificQuery, binding: ExecutionBinding) -> str:
    """Never includes the downstream objective, reliability weight or case lineage."""
    if query.outcome_mode not in {"continuous", "valid_readout", "attempted_experiment"}:
        raise ValueError("unknown prediction conditioning")
    if query.state.role not in {"historical", "predecision"}:
        raise ValueError("unknown state role")
    if query.state.role == "predecision" and not query.state.available_at:
        raise ValueError("predecision availability evidence required")
    for value in (query.state.content_sha256, query.state.feature_axis_sha256,
                  query.intervention_sha256, binding.checkpoint_sha256,
                  binding.sampling_indices_sha256):
        if not isinstance(value, str) or not re.fullmatch(r"[0-9a-f]{64}", value):
            raise ValueError("actual content hashes required")
    if not _finite(query.time_hours) or query.time_hours < 0 or type(binding.seed) is not int:
        raise ValueError("invalid execution setting")
    for value in (query.action_id, query.context, query.population, query.endpoint,
                  query.state.preprocessing_version, query.state.experimental_unit,
                  binding.input_transform_version, binding.runtime_version):
        if not isinstance(value, str) or not value.strip():
            raise ValueError("complete scientific and execution identity required")
    payload = json.dumps([asdict(query), asdict(binding)], sort_keys=True, allow_nan=False)
    return hashlib.sha256(payload.encode()).hexdigest()
