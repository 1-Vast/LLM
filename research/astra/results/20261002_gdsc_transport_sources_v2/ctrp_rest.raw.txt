function DTConnect(callback){
	var res = $.ajax({
		  url:  './rest/connect',
		  type: 'get',
		  context: document.body,
		  dataType: 'json',
		  async: true,
		  cache: false,
		  success: function(data) {
				var obj = (JSON.parse(res.responseText));
				callback(obj);
			},
		  error: function(){
			  removeWaitCursor();
			  alert('connect to server failed for DTConnect');
		  }
		});
}

function DTContact(from, subject, body, callback){
	var regObj = new Object();
	regObj.subject = subject;
	regObj.from = from;
	regObj.body = body;

	
	var res = $.ajax({
		  url:  './rest/contact',
		  type: 'post',
		  context: document.body,
		  data: JSON.stringify(regObj),
		  contentType: 'application/json',
		  async: true,
		  success: function(data) {	
				callback(data);
			},
		  error: function(){
			  removeWaitCursor();
			  alert('Contact message failed for DTContact');
		  }
		});
}

function DTGetCorrelation(compoundId, dataset, filter, callback){
	//console.log("Calling DTGetCorrelation with: " + dataset);
    var regObj = {cpd_id: compoundId,
    		dataset: dataset,
    		filter: filter};
	var res = $.ajax({
		  url:  './rest/geneAucCorrelation/',
		  type: 'post',
		  context: document.body,
          data: JSON.stringify(regObj),
		  dataType: 'json',
          cache: false,
          contentType: 'application/json',
		  async: true,
		  success: function(data) {
				callback(data);
			},
		  error: function(){
			  removeWaitCursor();
			  alert('ajax call failed for DTGetCorrelation');
		  }
		});
}

function DTGetCompoundCorrelation(featureName, dataset, filter, callback, errorCallback){
	//console.log("Calling DTGetCompoundCorrelation with: " + dataset);
	var regObj = {feature: featureName,
    		dataset: dataset,
    		filter: filter};
	var res = $.ajax({
		  url:  './rest/compoundAucCorrelation/',
		  type: 'post',
		  context: document.body,
          data: JSON.stringify(regObj),
		  dataType: 'json',
          cache: false,
          contentType: 'application/json',
		  async: true,
		  success: function(data) {
				callback(data);
			},
		  error: function(a,b,c,d){
			  removeWaitCursor();
			  if (errorCallback != null){
				  errorCallback(a,b,c,d, featureName, dataset);
			  } else {
				  alert('ajax call failed for DTGetCompoundCorrelation');
			  }
		  }
		});
}
