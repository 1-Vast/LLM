function getParameterByName(name) {
    name = name.replace(/[\[]/, "\\[").replace(/[\]]/, "\\]");
    var regex = new RegExp("[\\?&]" + name + "=([^&#]*)"),
        results = regex.exec(location.search);
    return results === null ? "" : decodeURIComponent(results[1].replace(/\+/g, " "));
}

function navigateToPage(pageId, callback, saveState){
	setDatasetFilterDisplay();
	
	$('.contentbody').hide(); // hide any of the full page bits
	// notify google analytics unless the load function will do it (with more than the default details)
	if (pageInitialization[pageId + ".reportFromLoadFunction"] != null){
		gaqReport(pageId);
	}
	
	
	//$(pageId).show();
	selectMenu(pageInitialization[pageId + ".menuClass"]);
	setFilterDisplay(pageInitialization[pageId + ".filterDisplay"]);
	
	$(pageId).show(100, 'swing', function(){
		if ((saveState != false)){
			navigationStack.push({page: pageId, callback: callback});
			try {
				window.history.pushState({page: pageId}, "ctd2 "+pageId, "?page="+pageId);
				
			} catch(ex){// swallow exceptions, just can't backup
				alert(ex);
			}
		}
		
		if (pageInitialization[pageId + ".html"] != null){
			try {
			$(pageId+"Body").load(pageInitialization[pageId + ".html"], function(){
				try {
					var loadFunction = (pageInitialization[pageId + ".loadFunction"]);
					if (loadFunction != null){
						(pageInitialization[pageId + ".loadFunction"]).call();
					}
				} catch(e) {
					alert(e);
					throw e;
				}
				if (callback != null){
					callback.call();
				}	
			} );
			pageInitialization[pageId + ".html"] = null; // this is done only once
			} catch (Ex){
				alert(Ex);
			}
			
		} else {
			// already initialized
			if (callback != null){
				callback.call();
			}
		}
		
		
	});
	
}

			dataArray.aoColumns[i].sTitle=colDisplayNames[i];
			dataArray.aoColumns[i].sName=colDisplayNames[i];
		}

		            "sSearch": "search all columns:",
		    	    "sLengthMenu": "show _MENU_ entries",
		    	    "sInfo": "showing _START_ to _END_ of _TOTAL_ entries"
		        },
		 		"aoColumns": dataArray.aoColumns,
				"aoColumnDefs": [
					 {
	                     // `data` refers to the data for the cell (defined by `mData`, which
	                     // defaults to the column being worked with, in this case is the first
	                     // Using `row[0]` is equivalent.
	                     "mRender": function ( data, type, row ) {
	                    	 var cnames  = row[4].split(';');
	                    	 var cids = row[5].split(';');
	                    	 
	                    	 var rowDisp = "";

	                    		 rowDisp = rowDisp + lnk;

	                    	 return rowDisp;
	                     },
	                     "aTargets": [ 4 ]
	                 },
	                 {
	                     // `data` refers to the data for the cell (defined by `mData`, which
	                     // defaults to the column being worked with, in this case is the first
	                     // Using `row[0]` is equivalent.

		             ],

function navigateBack(){
	try {
		navigationStack.pop(); //drop the current on the floor
		var previous = navigationStack.pop();
		if (previous != null){
			navigateToPage(previous.page, previous.callback, true);
		} 
	} catch(Exception){
		alert(Exception);
		// if it fails, just go back like we did nothing here
		// this gives backwards compatibility
	}
}


function fnCreateColumnSelect( aData )
{
    var r='<select><option value=""></option>', i, iLen=aData.length;
    for ( i=0 ; i<iLen ; i++ )
    {
        r += '<option value="'+aData[i]+'">'+aData[i]+'</option>';
    }
    return r+'</select>';
}

function collectFilterStrings(){
	var retVal = new Object();
	var subs = '';
	
	//var oTable;
	$('#dsCLSFilter input').each(function(k,v){
		if ($(v).is(':checked')){
			if ($(v).parent().data('filterOn') != null){
				if (subs.length>0)subs = subs + "\t";
				subs = subs + ($(v).parent().data('filterOn'));
			}
		}
	});
	var exclusions = '';
	$('#dsCLEFilter input').each(function(k,v){
		if ($(v).is(':checked')){
			if (exclusions.length>0)exclusions = exclusions + "\t";
			exclusions = exclusions + ($(v).parent().data('filterOn'));
		}
	});
	var datasets = '';
	$('#dsCLDFilter input').each(function(k,v){
		if ($(v).is(':checked')){
			if (datasets.length>0)datasets = datasets + "\t";
			datasets = datasets + ($(v).parent().data('filterOn'));
		}
	});
	
	retVal['site_primary'] = subs;
	retVal['growth_mode'] = exclusions;
	retVal['feature_dataset'] = datasets;
	
	return retVal;
}

 * Sets up a filter summary string to be the small version of the filters
 */
function setFilterSummary(){
	$('#filterSummarySite').html("");
	$('#filterSummaryMutation').html("");
	$('#filterSummaryGrowthMode').html("");
	var dsArrayChecked = new Array();
	var siteArrayChecked = new Array();
	var growthModeArrayChecked = new Array();
	var dsArrayUnChecked = new Array();
	var siteArrayUnChecked = new Array();
	var growthModeArrayUnChecked = new Array();
	
	var entireCategoriesChecked = new Array();
	$('#dsCLSFilter input').each(function(k,v){
		var wholeCategory = $(v).parent().data('isParent');
		// leep track of categories so in all lung types are checked then just say lun and don't enumerate them
		// this will fail if all sub cateforiues are checked but the top one isn't
		var name = ($(v).parent().data('display'));
		if ($(v).is(':checked')){
			var coveredByParent = false;
			$.each(entireCategoriesChecked, function(kk, cat){
				if (name.indexOf(cat) == 0) coveredByParent = true;
			});
			if (!coveredByParent) siteArrayChecked.push(name);
			if (wholeCategory) entireCategoriesChecked.push(name);
		} else {
			siteArrayUnChecked.push(name);
		}
	});
	$('#dsCLEFilter input').each(function(k,v){
		if ($(v).is(':checked')){
			growthModeArrayChecked.push(($(v).parent().data('display')));
		} else {
			growthModeArrayUnChecked.push(($(v).parent().data('display')));
		}
	});
	$('#dsCLDFilter input').each(function(k,v){
		if ($(v).is(':checked')){
			dsArrayChecked.push(($(v).parent().data('display')));
		} else {
			dsArrayUnChecked.push(($(v).parent().data('display')));	
		}
	});

	var str = getFilterSummaryString("mutations called by", dsArrayChecked, dsArrayUnChecked);
	$('#filterSummaryMutation').append(str);
	var str1 = getFilterSummaryString("growth mode", growthModeArrayChecked, growthModeArrayUnChecked);
	$('#filterSummaryGrowthMode').append(str1);
	var str2 = getFilterSummaryString("CCL", siteArrayChecked, siteArrayUnChecked);
	$('#filterSummarySite').append(str2);
	
	
}

function getFilterSummaryString(type, checked, unchecked){
	if (unchecked.length == 0) return "all "+ type + "s. ";
	if (unchecked.length == 1) return "excluding "+ unchecked[0]+". " ;
	if (checked.length <= 3) {
		var str = "";
		var count = 0;
		$.each(checked, function(k, item){
			if (item != null){
				if (count != 0) str = str + ", ";
				str = str + item;
				count++;
			}
		});
		return str + ". ";
	}
	if (checked.length > 3){
		return "multiple user-specified CCL" + + type + "s. ";
	}
	
}



var prevGeneFilter = "";
function filterDatasetColumns(){
	setFilterSummary();
	var filters = collectFilterStrings();
	var subs = filters['site_primary'];
	var exclusions = filters['growth_mode'];
	var datasets = filters['feature_dataset'];
	
	var currentPage = (lastElement(navigationStack)).page;
	if (currentPage == '#ctd2FeatureDetails'){
		// update the gene page 
		initializeDetailsForFeature($('#dsGeneNameFilter').val());
	} else if (currentPage == '#ctd2CompoundDetails'){
		// update the compound page
		initializeCompoundDetailsForCompound($('#dsCompoundIdFilter').val());
	} 
	// update the features table
	// the fnFilter is not called if empty so we fake it out like so
	//var featureDetailTable = $('#datasetTableTable').dataTable();
	
	if (subs.length == 0) subs = "\t";
	if (exclusions.length == 0) exclusions = "\t";
	if (datasets.length == 0) datasets = "\t";
	
	var featureName = $('#dsGeneNameFilter').val();
	if (prevGeneFilter != featureName){
		//featureDetailTable.fnFilter( featureName, 4);
		prevGeneFilter = featureName;
	}		
	//featureDetailTable.fnMultiFilter( { "Cell Line Subset": subs, "Cell Line Exclusion": exclusions, "Dataset":datasets });
	
	var featureSummaryTable = $('#summaryDatasetTableTable').dataTable();
	
	
	// create the list of unique 'sub excl dataset' and then join them using '\t' into one string
	// in this the sub, exxcl and ds are seperated by spaces and each set is seperated by a tab
	var filterArrays = collectFilterArrays();
	var subArray = filterArrays['site_primary_array'];
	var exclArray = filterArrays['growth_mode_array'];
	var dsArray = filterArrays['dataset_array'];
	var uniqueEntries = "\t";
	$.each(subArray, function(k, sub){
		$.each(exclArray, function(k, excl){
			$.each(dsArray, function(k, ds){
				uniqueEntries = uniqueEntries + sub + " " + excl + " " + ds + "\t";
			});		
		});
	});
	
	var warningMsg = "";
	if (subArray.length == 0) {
		warningMsg +="\nPlease choose at least one primary site/subtype.";
	} 
	if (exclArray.length == 0) {
		warningMsg +="\nPlease choose at least one growth mode.";
	} 
	if (dsArray.length == 0) {
		warningMsg +="\nPlease choose at least one Dataset.";
	}
	
	if (warningMsg.length == 0){
		if (allFiltersAreChecked()){
			featureSummaryTable.fnMultiFilter( { "Entries": "" });
	
		} else {
			featureSummaryTable.fnMultiFilter( { "Entries": uniqueEntries });
		}
		//removeWaitCursor();
	} else {
		removeWaitCursor();
		alert(warningMsg);
		
	}
}

$.fn.dataTableExt.oApi.fnMultiFilter = function( oSettings, oData ) {
	if (oSettings == null) return;
    for ( var key in oData )
    {
        if ( oData.hasOwnProperty(key) )
        {
            for ( var i=0, iLen=oSettings.aoColumns.length ; i<iLen ; i++ )
            {
                if( oSettings.aoColumns[i].sName == key )
                {
                    /* Add single column filter */
                    oSettings.aoPreSearchCols[ i ].sSearch = oData[key];
                    break;
                }
            }
        }
    }
    this.oApi._fnDraw( oSettings );
   
};


function daysBetween( date1, date2 ) {
	  //Get 1 day in milliseconds
	  var one_day=1000*60*60*24;

	  // Convert both dates to milliseconds
	  var date1_ms = date1.getTime();
	  var date2_ms = date2.getTime();

	  // Calculate the difference in milliseconds
	  var difference_ms = date2_ms - date1_ms;
	    
	  // Convert back to days and return
	  return Math.round(difference_ms/one_day); 
	};



$.extend({

      var vars = [], hash;
      var hashes = window.location.href.slice(window.location.href.indexOf('?') + 1).split('&');
      for(var i = 0; i < hashes.length; i++)
      {
        hash = hashes[i].split('=');
        vars.push(hash[0]);
        vars[hash[0]] = hash[1];
      }
      return vars;
    },
    getUrlVar: function(name){
      return $.getUrlVars()[name];
    }
  });
