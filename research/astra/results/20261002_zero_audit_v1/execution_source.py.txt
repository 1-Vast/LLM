"""Trace frozen EGR1 zeros without changing STATE inference or utility.

All statistics concern fixed historical control pools and model-internal
resampling. No seed, output position, or assay plate is a physical replicate.
"""

from contextlib import contextmanager
from datetime import datetime, timezone
from itertools import combinations
import json
from pathlib import Path
import subprocess
import sys
import time

import anndata as ad
import numpy as np

from research.astra import paired_state
from tools.datasets.state_prospective_input import CONTROL, KEY, PERT, digest, write_json


def endpoint_index(axis):
    if len(axis) != 2000 or axis.count("0546:EGR1") != 1 or axis[546] != "0546:EGR1":
        raise ValueError("EGR1 identity unavailable; never substitute zero")
    return 546


def numbers(values):
    values = np.asarray(values)
    if values.ndim != 1 or not len(values) or not np.isfinite(values).all():
        raise ValueError("missing or invalid readout; never substitute zero")
    return dict(values=values.tolist(), float_hex=[float(v).hex() for v in values],
                dtype=str(values.dtype), sha256=paired_state.tensor_digest(values),
                minimum=float(values.min()), maximum=float(values.max()),
                exact_zeros=int((values == 0).sum()), negative=int((values < 0).sum()),
                native_mean=float(values.mean()), mean_float64=float(values.mean(dtype=np.float64)))


def choice(utility, tolerance=1e-6):
    utility = np.asarray(utility, dtype=float)
    if utility.shape != (3,) or not np.isfinite(utility).all():
        raise ValueError("complete finite three-action utility required")
    best = np.flatnonzero(utility.max() - utility <= tolerance).tolist()
    gap = float(np.sort(utility)[-1] - np.sort(utility)[-2])
    return dict(candidate_indices=best, selected=best[0] if len(best) == 1 else None,
                top_gap=gap, reason="exact_numeric_top_tie" if gap == 0 else
                "numeric_tolerance_tie" if len(best) > 1 else "unique_numeric_maximum")


def paired_contrasts(utilities, pairing_verified, tolerance=1e-6):
    utilities = np.asarray(utilities, dtype=float)
    if not pairing_verified:
        raise ValueError("actual within-request paired basals required")
    if utilities.shape != (3, 3) or not np.isfinite(utilities).all():
        raise ValueError("three frozen numeric draws and three actions required")
    result = []
    for a, b in combinations(range(3), 2):
        differences = utilities[:, a] - utilities[:, b]
        mean = float(differences.mean())
        every_tied = bool(np.all(np.abs(differences) <= tolerance))
        result.append(dict(actions=[a, b], utility_differences=differences.tolist(),
                           mean_difference=mean,
                           descriptive_sd_over_sqrt_K=float(differences.std(ddof=1) / np.sqrt(3)),
                           MC_SE_status="conditional on independent numeric resampling; no validated interval",
                           every_seed_within_numeric_tolerance=every_tied,
                           fixed_pair_convexity_pass=not every_tied or abs(mean) <= tolerance,
                           practical_advantage="unknown", practical_equivalence="unknown"))
    return result


@contextmanager
def activation_observer(model_class, records, endpoint=546):
    """Read-only ReLU hook; restores both class and module hooks on failure."""
    original = model_class.predict_step

    def observe(model, batch, *args, **kwargs):
        captured = []

        def capture(module, inputs, output):
            captured.append((inputs[0].detach().cpu().numpy().copy(),
                             output.detach().cpu().numpy().copy()))

        handle = model.relu.register_forward_hook(capture)
        try:
            output = original(model, batch, *args, **kwargs)
        finally:
            handle.remove()
        if len(captured) != 1:
            raise ValueError("exactly one observed final ReLU required")
        before, after = captured[0]
        returned = output["preds"].detach().cpu().numpy()
        if before.shape != after.shape or before.shape[-1] <= endpoint:
            raise ValueError("activation feature shape changed")
        if not np.isfinite(before).all() or not np.isfinite(after).all():
            raise ValueError("nonfinite activation")
        if not np.array_equal(after, np.maximum(before, 0)) or not np.array_equal(after.reshape(returned.shape), returned):
            raise ValueError("returned prediction differs from observed ReLU output")
        records.append(dict(action=batch["pert_name"][0], rows=len(returned),
                            actual_basal_sha256=paired_state.tensor_digest(batch["ctrl_cell_emb"]),
                            before_sha256=paired_state.tensor_digest(before),
                            returned_sha256=paired_state.tensor_digest(returned),
                            pre_EGR1=numbers(before.reshape(-1, before.shape[-1])[:, endpoint]),
                            post_EGR1=numbers(returned[:, endpoint]),
                            relu_exact=True, returned_equals_observed=True))
        return output

    model_class.predict_step = observe
    try:
        yield
    finally:
        model_class.predict_step = original


def worker(cell, out):
    from state.tx.models.state_transition import StateTransitionPerturbationModel

    contract = json.loads((cell / "contract.json").read_text(encoding="utf-8"))
    receipt = json.loads((cell / "receipt.json").read_text(encoding="utf-8"))
    records, error = [], None
    original = StateTransitionPerturbationModel.predict_step
    try:
        with activation_observer(StateTransitionPerturbationModel, records, endpoint_index(contract["axis"])):
            paired_state.worker(cell / "query.h5ad", out / "prediction.h5ad", cell / "contract.json",
                                receipt["actions"], receipt["count"], receipt["seed"])
    except Exception as exc:
        error = f"{type(exc).__name__}: {exc}"
        raise
    finally:
        write_json(out / "activation_trace.json", dict(records=records, error=error,
                   class_hook_restored=StateTransitionPerturbationModel.predict_step is original))


def frozen_rows(root):
    grid = root / "research/astra/results/20261002_paired_matrix_v1"
    manifest = json.loads((grid / "manifest.json").read_text(encoding="utf-8"))
    for name, expected in manifest.items():
        if digest(grid / name) != expected:
            raise ValueError(f"frozen grid artifact changed: {name}")
    summary = json.loads((grid / "summary.json").read_text(encoding="utf-8"))
    for row in summary["rows"]:
        cell = Path(row["directory"])
        for name, field in (("receipt.json", "receipt_sha256"), ("request_predictions.npy", "prediction_sha256")):
            if digest(cell / name) != row[field]:
                raise ValueError(f"frozen source changed: {cell / name}")
        source_manifest = json.loads((cell / "manifest.json").read_text(encoding="utf-8"))
        for name, expected in source_manifest.items():
            if digest(cell / name) != expected:
                raise ValueError(f"source manifest mismatch: {cell / name}")
    return summary


def audit_cell(row, actions):
    cell = Path(row["directory"])
    contract = json.loads((cell / "contract.json").read_text(encoding="utf-8"))
    index = endpoint_index(contract["axis"])
    query, prediction = ad.read_h5ad(cell / "query.h5ad"), ad.read_h5ad(cell / "prediction.h5ad")
    trace = json.loads((cell / "paired_trace.json").read_text(encoding="utf-8"))
    if not query.obs.equals(prediction.obs):
        raise ValueError("query/response identity changed")
    baseline = query[query.obs["role"].astype(str) == "baseline_control"].copy()
    sampled, indices, ids = paired_state.sample_baseline(baseline, 16, row["seed"])
    if trace["source_indices_zero_based"] != indices or trace["source_sample_ids"] != ids:
        raise ValueError("source identity or sampling changed")
    calls = [c for c in trace["calls"] if c["paired"]]
    if len(calls) != 3 or {c["action"] for c in calls} != set(actions):
        raise ValueError("incomplete executed menu")
    for call in calls:
        if (call["basal_sha256"] != paired_state.tensor_digest(sampled) or
                call["source_sample_ids"] != ids or call["source_indices_zero_based"] != indices):
            raise ValueError("unpaired actual action inputs")
    matrix = np.asarray(prediction.obsm[KEY])
    mask = query.obs["role"].astype(str) == "prediction_request"
    saved = np.load(cell / "request_predictions.npy")
    if matrix.shape != (78, 2000) or saved.shape != (48, 2000) or not np.array_equal(saved, matrix[mask]):
        raise ValueError("saved matrix differs from returned predictions")
    details, scores = [], []
    for i, action in enumerate(actions):
        values = matrix[np.asarray(query.obs[PERT].astype(str) == action), index]
        if len(values) != 16:
            raise ValueError("frozen query count changed")
        readout = numbers(values)
        if readout["native_mean"] != row["endpoint_scores"][i]:
            raise ValueError("original endpoint mean changed")
        scores.append(-readout["native_mean"])
        details.append(dict(action=action, returned_EGR1=readout,
                            treatment_minus_shared_input_set=numbers(values.astype(float) - sampled[:, index]),
                            difference_scope="computational positions in set-to-set model, not cell trajectories",
                            difference_of_set_means=readout["mean_float64"] - float(sampled[:, index].mean(dtype=np.float64)),
                            utility=-readout["native_mean"], transform="negative native float32 mean; no new threshold"))
    return dict(pool=row["pool"], seed=row["seed"], source_directory=str(cell),
                paired_inputs_verified=True, sampled_source_ids=ids,
                original_baseline_EGR1=numbers(np.asarray(baseline.obsm[KEY])[:, index]),
                sampled_baseline_EGR1=numbers(sampled[:, index]),
                returned_control_EGR1=numbers(matrix[np.asarray(query.obs[PERT].astype(str) == CONTROL), index]),
                returned_control_scope="independently sampled model control; unpaired with action basal",
                actions=details, utilities=scores, choice=choice(scores))


def policy_diagnostics(rows):
    results = {}
    for pool in sorted({r["pool"] for r in rows}):
        selected = [r for r in rows if r["pool"] == pool]
        utilities = np.asarray([r["utilities"] for r in selected])
        aggregate = choice(utilities.mean(axis=0))
        contrasts = paired_contrasts(utilities, all(r["paired_inputs_verified"] for r in selected))
        results[pool] = dict(seeds=[r["seed"] for r in selected],
                            estimand="uniform mean of negative predicted set-mean log1p EGR1 over frozen basal resampling",
                            randomness="with-replacement numerical resampling of fixed 30-cell historical control pool",
                            single_seed=[dict(seed=r["seed"], **r["choice"]) for r in selected],
                            fixed_K_aggregate=aggregate, aggregate_utilities=utilities.mean(axis=0).tolist(),
                            paired_contrasts=contrasts,
                            uncertainty_preserving=dict(candidate_indices=[0, 1, 2], selected=None,
                            reason="practical delta, validated biological intervals and measured outcome matching unavailable",
                            numeric_candidate_indices=aggregate["candidate_indices"]),
                            fixed_action=dict(selected=0, reason="predeclared low-dose comparator; not asserted optimal"),
                            prospective_inference_forwards=dict(single_seed=4, fixed_K_aggregate=12,
                                                               uncertainty_preserving=12, fixed_action=0),
                            comparison_scope="post-hoc diagnostic using previously inspected outputs; not independent validation",
                            realized_utility=None, refusal_contribution=None, physical_CI=None,
                            meaningful_delta=None, physical_attempts=0, state_cost=None,
                            next_step="validate EGR1 readout and matched measured action outcomes; more seeds cannot repair model bias")
    return results


def run(root, out, observe=False):
    root, out = root.resolve(), out.resolve()
    out.mkdir(parents=True, exist_ok=False)
    summary = frozen_rows(root)
    source = root / "data/external/arc_state/source/src/state/tx/models/state_transition.py"
    frozen = {str(Path(__file__).resolve()): digest(__file__), str(source): digest(source),
              str(Path(paired_state.__file__).resolve()): digest(paired_state.__file__)}
    for row in summary["rows"]:
        cell = Path(row["directory"])
        for name in ("contract.json", "query.h5ad", "prediction.h5ad", "paired_trace.json", "request_predictions.npy"):
            frozen[str(cell / name)] = digest(cell / name)
    write_json(out / "freeze.json", dict(frozen_at_utc=datetime.now(timezone.utc).isoformat(),
               inputs=frozen, source_summary_sha256=digest(root / "research/astra/results/20261002_paired_matrix_v1/summary.json"),
               observed_previously=True, new_seeds=[], observation_requests=9 if observe else 0,
               observation_forward_budget=36 if observe else 0, count_per_action=16,
               actions=summary["action_labels"], seeds=[17, 42, 103], utility=summary["utility"],
               numerical_tolerance=1e-6, meaningful_delta=None,
               observation="read-only ReLU hook; compare full output exactly; no changed prediction policy",
               physical_attempts=0, statistical_intervals="not_reported", sequential_stopping="not_used"))
    (out / "execution_source.py.txt").write_bytes(Path(__file__).read_bytes())
    (out / "state_transition.py.txt").write_bytes(source.read_bytes())
    rows, replays = [], []
    started = time.perf_counter()
    try:
        for row in summary["rows"]:
            rows.append(audit_cell(row, summary["action_labels"]))
            write_json(out / "readout_audit.json", rows)
        write_json(out / "policy_diagnostics.json", policy_diagnostics(rows))
        if observe:
            for row in summary["rows"]:
                cell = Path(row["directory"])
                replay = out / f"{row['pool']}_seed{row['seed']}"
                replay.mkdir()
                command = [sys.executable, "-m", "research.astra.zero_audit", "worker", "--cell", str(cell), "--out", str(replay)]
                tick = time.perf_counter()
                done = subprocess.run(command, cwd=root, capture_output=True, text=True,
                                      encoding="utf-8", errors="replace", timeout=600)
                (replay / "stdout.txt").write_text(done.stdout, encoding="utf-8")
                (replay / "stderr.txt").write_text(done.stderr, encoding="utf-8")
                result = dict(pool=row["pool"], seed=row["seed"], command=command,
                              returncode=done.returncode, elapsed_seconds=time.perf_counter() - tick,
                              full_output_bitwise_equal=False)
                replays.append(result)
                write_json(out / "replay_receipts.json", replays)
                if done.returncode:
                    raise RuntimeError("observation replay failed; retained failure receipt")
                old = ad.read_h5ad(cell / "prediction.h5ad")
                new = ad.read_h5ad(replay / "prediction.h5ad")
                result["full_output_bitwise_equal"] = bool(old.obs.equals(new.obs) and
                    old.obsm[KEY].shape == new.obsm[KEY].shape and old.obsm[KEY].dtype == new.obsm[KEY].dtype and
                    paired_state.tensor_digest(old.obsm[KEY]) == paired_state.tensor_digest(new.obsm[KEY]))
                if not result["full_output_bitwise_equal"]:
                    raise ValueError("observation replay differs; old zero cause remains uncertified")
                observed = json.loads((replay / "activation_trace.json").read_text(encoding="utf-8"))
                if observed["error"] or not observed["class_hook_restored"] or len(observed["records"]) != 4:
                    raise ValueError("incomplete activation receipt")
                for call in observed["records"]:
                    exported = np.asarray(new.obsm[KEY])[np.asarray(new.obs[PERT].astype(str) == call["action"]), 546]
                    if numbers(exported)["sha256"] != call["post_EGR1"]["sha256"]:
                        raise ValueError("export altered EGR1 after native forward")
                result["exported_EGR1_equals_native_forward"] = True
                zero_calls = [c for c in observed["records"] if c["action"] != CONTROL and c["post_EGR1"]["exact_zeros"] == 16]
                result["all_zero_actions"] = len(zero_calls)
                result["all_zero_actions_with_every_pre_value_negative"] = sum(c["pre_EGR1"]["negative"] == 16 for c in zero_calls)
                write_json(out / "replay_receipts.json", replays)
                print(f"observed {len(replays)}/9: {row['pool']} seed{row['seed']}; exact full replay", flush=True)
        if any(digest(path) != expected for path, expected in frozen.items()):
            raise ValueError("frozen input/source changed during audit")
        receipt = dict(valid=True, audited_requests=len(rows),
                       zero_action_means=sum(a["returned_EGR1"]["native_mean"] == 0 for r in rows for a in r["actions"]),
                       exact_numeric_top_tie_refusals=sum(r["choice"]["reason"] == "exact_numeric_top_tie" for r in rows),
                       newly_executed_requests=len(replays), newly_executed_forwards=len(replays) * 4,
                       exact_full_replays=sum(r["full_output_bitwise_equal"] for r in replays),
                       zero_actions_traced_to_strictly_negative_pre_relu=sum(r.get("all_zero_actions_with_every_pre_value_negative", 0) for r in replays),
                       computational_stability="exact replay of same inputs" if observe else "not_newly_tested",
                       meaningful_action_advantage="unknown", experimental_benefit="not_identified",
                       physical_attempts=0, independent_cultures="unknown", actual_cost="unknown",
                       elapsed_seconds=time.perf_counter() - started, input_hashes_unchanged=True)
    except Exception as exc:
        receipt = dict(valid=False, error=f"{type(exc).__name__}: {exc}", completed_observation_requests=len(replays),
                       elapsed_seconds=time.perf_counter() - started, physical_attempts=0)
        raise
    finally:
        write_json(out / "receipt.json", receipt)
        write_json(out / "manifest.json", {p.relative_to(out).as_posix(): digest(p) for p in out.rglob("*") if p.is_file()})
    print(json.dumps(receipt), flush=True)


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("mode", choices=["run", "worker"])
    parser.add_argument("--out", type=Path, required=True)
    parser.add_argument("--root", type=Path, default=Path.cwd())
    parser.add_argument("--cell", type=Path)
    parser.add_argument("--observe", action="store_true")
    args = parser.parse_args()
    if args.mode == "worker":
        worker(args.cell, args.out)
    else:
        run(args.root, args.out, args.observe)
