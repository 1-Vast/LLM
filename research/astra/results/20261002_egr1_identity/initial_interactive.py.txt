# Exact Python body of the earlier interactive read-only check.
# This earlier check materialized the CSR arrays; verify_egr1.py is the chunked reproduction.
import h5py,numpy as np,json,hashlib
from scipy.sparse import csr_matrix
p='data/external/arc_state/tahoe_metadata_source/c39.h5ad'
identity=json.load(open('data/external/arc_state/hvg_identity.json'))
print('identity method',identity['method'])
print('identity normalization',identity['normalization'])
with h5py.File(p,'r') as f:
 names=f['var/gene_name'].asstr()[:]
 idx=np.flatnonzero(names=='EGR1')
 print('EGR1 original column(s)',idx.tolist(),'identity expected',identity['gene_indices'][546])
 g=f['X']; x=csr_matrix((g['data'][:],g['indices'][:],g['indptr'][:]),shape=tuple(g.attrs['shape']))
 raw=x[:,int(idx[0])].toarray().ravel()
 observed=f['obsm/X_hvg'][:,546]
 derived=np.log1p(raw)
 diff=np.abs(derived-observed)
 print(json.dumps({'rows':len(raw),'raw_nonzero':int(np.count_nonzero(raw)), 'raw_min':float(raw.min()),'raw_max':float(raw.max()),'hvg_min':float(observed.min()),'hvg_max':float(observed.max()),'max_abs_log1p_difference':float(diff.max()),'n_difference_above_1e5':int(np.count_nonzero(diff>1e-5)),'mean_hvg':float(observed.mean())},indent=2))
for q in [p,'data/virtual_cell/tahoe_c39_x_hvg_feature_names.json','data/external/arc_state/hvg_identity.json','data/external/arc_state/source/src/state/tx/models/state_transition.py']:
 with open(q,'rb') as s: print(q,hashlib.file_digest(s,'sha256').hexdigest())
