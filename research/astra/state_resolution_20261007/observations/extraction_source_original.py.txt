read=lambda p:json.loads(p.read_text());sha=lambda p:hashlib.sha256(p.read_bytes()).hexdigest()
def write(p,x):
 if p.exists():raise FileExistsError(p)
 p.write_text(json.dumps(x,indent=2)+'\n')
start=time.perf_counter();plan=read(OLD/'world/metadata_freeze.json');tech=read(OLD/'world/technical_screen_freeze.json');contract=read(OLD/'world/contract.json');byid={x['condition_id']:x for x in tech['rows']}

write(OUT/'ENDPOINT_PROTOCOL.json',protocol);write(OUT/'ENDPOINT_FREEZE.json',{'protocol_sha256':sha(OUT/'ENDPOINT_PROTOCOL.json'),'frozen_at_utc':datetime.now(timezone.utc).isoformat(),'corrected_evaluation_values_read':False})

   delta=x.mean(0)-c.mean(0);raw=float(np.mean(delta**2));vt=float(np.mean(x.var(0,ddof=1)));vc=float(np.mean(c.var(0,ddof=1)));corrected=raw-vt/len(x)-vc/len(c)
   result={k:condition[k] for k in ['condition_id','drug','dose_uM','plate','cell','time_hours','assay','split','chemical_group']};result.update(role=role,raw_squared_response=raw,signed_effect=corrected,treated_n=len(x),control_n=len(c),treated_mean_gene_variance=vt,control_mean_gene_variance=vc,treated_noise_term=vt/len(x),control_noise_term=vc/len(c),subset_sha256=hashlib.sha256(json.dumps([str(ids[i]) for i in indices]).encode()).hexdigest(),control_subset_sha256=hashlib.sha256(json.dumps([str(ids[i]) for i in plan['controls'][condition['plate']]['reference_rows']]).encode()).hexdigest());rows.append(result)
   if role=='full':
    full.append(result);order=sorted(range(len(x)),key=lambda i:hashlib.sha256(('corrected-crossproduct-v1:'+str(ids[indices[i]])).encode()).hexdigest());a,b=x[order[::2]],x[order[1::2]];ca,cb=c[::2],c[1::2];cross.append(float(np.mean((a.mean(0)-ca.mean(0))*(b.mean(0)-cb.mean(0)))))
