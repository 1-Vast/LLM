import json, sys, time, statistics, hashlib, platform
from pathlib import Path
root = Path.cwd()
sys.path.insert(0, str(root / 'src'))
from research.dual_core_live.benchmark import fixed_action, best_action
from agent.model_audit import ModelAuditAgent
out = root / 'outputs/dual_core_live_20261001'
freeze = out / 'fixed_benchmark_v1/predeclared.json'
cards = json.loads(freeze.read_text())['cards']
auditor = ModelAuditAgent()
functions = {'fixed_order': lambda c: fixed_action(c), 'one_step_optimizer': lambda c: best_action(c), 'report_only_auditor': lambda c: auditor.review(c, {'action': best_action(c), 'terminal_authorized': False})}
rows = {}
for name, function in functions.items():
    for c in cards: function(c)
    trials = []
    for _ in range(3):
        start = time.perf_counter()
        for _ in range(1000):
            for c in cards: function(c)
        trials.append((time.perf_counter()-start)/(1000*len(cards)))
    rows[name] = {'calls_per_trial': 1000*len(cards), 'seconds_per_card_trials': trials, 'median_seconds_per_card': statistics.median(trials)}
record = {'scope': 'Local warmed engineering timing on the fixed contract cards; not independent biological observations, cold start, network throughput or STATE speed.', 'environment': {'interpreter': sys.executable, 'python': sys.version, 'platform': platform.platform()}, 'frozen_input_sha256': hashlib.sha256(freeze.read_bytes()).hexdigest(), 'source_sha256': {p: hashlib.sha256((root/p).read_bytes()).hexdigest() for p in ('research/dual_core_live/benchmark.py','src/agent/model_audit.py')}, 'rows': rows}
with (out/'local_timing.json').open('x',encoding='utf-8') as f: json.dump(record,f,indent=2,allow_nan=False)
print(json.dumps(rows,indent=2))
