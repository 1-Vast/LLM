from pathlib import Path
import json, hashlib, shutil, sys
root=Path.cwd();src=root/'outputs/dual_core_live_20261001';dest=root/'outputs/dual_core_live_publication_20261001';dest.mkdir(exist_ok=False)
rows=[]
for p in sorted(src.rglob('*')):
    if not p.is_file():continue
    rel=p.relative_to(src); reason=None
    if 'test_temp' in rel.parts:reason='generated_test_fixture_local_only'
    elif '__pycache__' in rel.parts or p.suffix in ('.pyc','.pyo'):reason='python_cache_local_only'
    elif p.name.startswith('.env'):reason='configuration_excluded'
    sha=hashlib.sha256(p.read_bytes()).hexdigest()
    if reason is None:
        target=dest/rel;target.parent.mkdir(parents=True,exist_ok=True);shutil.copyfile(p,target)
        if hashlib.sha256(target.read_bytes()).hexdigest()!=sha:raise RuntimeError('copy_mismatch')
    rows.append({'original':str(p.relative_to(root)),'original_sha256':sha,'published':str(rel).replace('\\','/') if reason is None else None,'exclusion':reason})
with (dest/'curation_receipt.json').open('x',encoding='utf-8') as f:json.dump({'command':[sys.executable,'-c','curation command retained alongside publication'],'files':rows,'originals_unchanged':True},f,indent=2)
print(json.dumps({'published':sum(r['published'] is not None for r in rows),'excluded_local_only':sum(r['published'] is None for r in rows)}))
